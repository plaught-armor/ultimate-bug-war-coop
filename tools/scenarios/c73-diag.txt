# Card 073 step 1: where the grenade count lives. The client takes a frag pickup, fires, throws; then the host
# takes one, fires, throws. Both log the item inventory and grenade ability specs (UBWCOOP_C73_LOG=1).
# Run: SCENARIO=tools/scenarios/c73-diag.txt HOST_ENV="UBWCOOP_AUTODROPSHIP=1 UBWCOOP_C73_LOG=1" \
#      CLIENT_ENV="UBWCOOP_C73_LOG=1" run65.sh <tag>
host wait state pawn=host-own timeout=240
client wait state pawn=client-own timeout=120
host ui continue
sleep 3
client shot start
client goto actor BP_GrenadeAmmoPickup_Frag_C 80
sleep 3
client shot after-pickup
client look 0 -30
client key LeftMouseButton 800
sleep 2
client shot after-shot
client key G
sleep 4
client shot after-throw
host goto actor BP_GrenadeAmmoPickup_Frag_C 80
sleep 3
host shot after-pickup
host look 0 -30
host key LeftMouseButton 800
sleep 2
host key G
sleep 4
host shot after-throw
sleep 2
client quit
sleep 2
host quit

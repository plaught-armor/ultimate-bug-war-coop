# Card 072: the client owns its ammunition counts and its server copy mirrors them.
# The client walks first and takes both BP_AmmoPickup_MoritaSmall_C on the way to the Morita (the copy reaches 240),
# fires two full magazines with a reload after each, then walks to a third pickup: the copy must take it.
# Run: SCENARIO=tools/scenarios/c72-mirror.txt HOST_ENV="UBWCOOP_AUTODROPSHIP=1 UBWCOOP_C69_LOG=1" \
#      CLIENT_ENV="UBWCOOP_C69_LOG=1" run65.sh <tag>

client wait state pawn=client-own timeout=120
host ui continue
sleep 3
client goto actor BP_WeaponPickup_MoritaMk1_C 120
client look actor BP_WeaponPickup_MoritaMk1_C
client key E
host wait log "card021: host GiveWeaponByClass\(BP_WeaponDefinition_MoritaMk1_C.*server-copy" timeout=15
client key Two
client wait log "switch to weapon 1: Equip sent" timeout=15
sleep 3
client shot equipped
client look 0 -45
client key LeftMouseButton 6000
sleep 1
client key R
sleep 4
client key LeftMouseButton 6000
sleep 1
client key R
sleep 4
client shot before-third
client goto actor BP_AmmoPickup_MoritaLarge_C 100
host wait log "card071: host .*on server copy" count=3 timeout=20
sleep 3
client shot after-third
sleep 2
client quit
sleep 2

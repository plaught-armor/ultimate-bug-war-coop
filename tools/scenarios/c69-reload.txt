# Card 069: the client's reserve ammo and reload (log only).
# Both players pick up the Morita, equip it, fire part of a magazine at the ground, press reload, and take a shot.
# Run: SCENARIO=tools/scenarios/c69-reload.txt HOST_ENV="UBWCOOP_AUTODROPSHIP=1 UBWCOOP_C69_LOG=1" \
#      CLIENT_ENV="UBWCOOP_C69_LOG=1" run65.sh <tag>
host wait state pawn=host-own timeout=240
client wait state pawn=client-own timeout=120
host ui continue
sleep 3
# The client goes first: the two BP_AmmoPickup_MoritaSmall_C on the way are overlap pickups, and the first pawn
# through takes them (in c69a the host took both, 120 rounds each).
client goto actor BP_WeaponPickup_MoritaMk1_C 120
client look actor BP_WeaponPickup_MoritaMk1_C
client key E
host wait log "card021: host GiveWeaponByClass\(BP_WeaponDefinition_MoritaMk1_C.*server-copy" timeout=15
host goto actor BP_WeaponPickup_MoritaMk1_C 120
host look actor BP_WeaponPickup_MoritaMk1_C
host key E
host wait log "card021: host GiveWeaponByClass\(BP_WeaponDefinition_MoritaMk1_C.*host-own" timeout=15
client key Two
client wait log "switch to weapon 1: Equip sent" timeout=15
host key Two
sleep 3
host shot equipped
client shot equipped
# Fire at the ground in front, then reload.
host look 0 -45
client look 0 -45
host key LeftMouseButton 1500
sleep 2
host key R
sleep 4
host shot after-reload
client key LeftMouseButton 1500
sleep 2
client shot after-fire
client key R
sleep 4
client shot after-reload
sleep 2
client quit
sleep 2
host quit

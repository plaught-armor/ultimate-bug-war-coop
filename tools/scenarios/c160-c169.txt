# Cards 160 and 169, one run: the client's ADS zoom through the host's server copy (card 160), and the held visuals
# an AbilityTask SpawnActor makes (card 169: frag, incendiary, knife; flare last), with the host's own throws, melee
# and ADS as the controls. One FAST=1 pair run on FED_C1M1 (GOD=0). Predictions in docs/cards/160-client-ads-zoom.md
# and docs/cards/169-ability-spawn-visuals.md. Every state step waits for its log line.
# Run: SCENARIO=tools/scenarios/c160-c169.txt DEPLOY=1 FAST=1 GOD=0
#      HOST_EXTRA_ENV="UBWCOOP_AUTODROPSHIP=1" tools/run_pair.sh <tag> 900
host wait state pawn=host-own timeout=240
client wait state pawn=client-own timeout=120
host ui continue
host wait log "playlog: host t=\d+ clock 1 " timeout=30
client wait log "playlog: client t=\d+ clock 1:" timeout=10
# Card 169 control: the host takes the second grenade pair (13392 13337, 13368 13374) and throws its own frag, then
# its own incendiary. The destroy line gives socket, visibility and camera distance.
host mark
host warp actor point 13250 13350 250
host goto actor BP_GrenadeAmmoPickup_Frag_C_1 60 timeout=20
host goto actor BP_GrenadeAmmoPickup_Incendiary_C_1 60 timeout=20
host wait log "card095: host t=\d+ collision BP_GrenadeAmmoPickup_Frag_C_1 .* 1 -> 0" since-mark timeout=10
host wait log "card095: host t=\d+ collision BP_GrenadeAmmoPickup_Incendiary_C_1 .* 1 -> 0" since-mark timeout=10
sleep 2
host look point 13250 30000 900
host mark
host key G
host wait log "card169: host t=\d+ held .* destroyed after " since-mark timeout=8
sleep 4
host look point 14250 30000 900
host mark
host key G
host wait log "card169: host t=\d+ held .* destroyed after " since-mark timeout=8
sleep 3
# The client takes the Morita at 6445 11469; the first grenade pair lies on the way (6453 11426, 6466 11451), so
# its pickup lines come before the ADS mark and the grenade waits below read the whole log (r1 fixed).

sleep 2

client wait log "card169: client t=\d+ held .* destroyed after " since-mark timeout=8
host wait log "card158: host t=\d+ server copy throw " since-mark timeout=10
sleep 4

client mark
